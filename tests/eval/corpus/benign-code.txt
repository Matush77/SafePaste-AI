import { useEffect, useState } from "react";

export function SearchBox({ onSearch }) {
  const [query, setQuery] = useState("");

  useEffect(() => {
    const id = setTimeout(() => onSearch(query), 300);
    return () => clearTimeout(id);
  }, [query, onSearch]);

  return <input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Search" />;
}

=== code/sql-query
SELECT o.id, o.created_at, c.country, SUM(li.quantity * li.unit_price) AS total
FROM orders o
JOIN customers c ON c.id = o.customer_id
JOIN line_items li ON li.order_id = o.id
WHERE o.created_at >= '2024-01-01' AND o.status IN ('paid', 'shipped')
GROUP BY o.id, o.created_at, c.country
HAVING SUM(li.quantity * li.unit_price) > 1000
ORDER BY total DESC
LIMIT 50;
